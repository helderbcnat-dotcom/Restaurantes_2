Execute no SQL Editor (se ainda não fez): -- Execute no SQL Editor do Supabase para configurar a tabela: CREATE TABLE IF NOT EXISTS pedidos ( id bigserial PRIMARY KEY, user_id uuid REFERENCES auth.users NOT NULL, restaurante text NOT NULL, prato text, preco numeric, nota numeric, data date, comentario text, foto text, nota_fiscal_foto text, criado_em timestamptz DEFAULT now() ); ALTER TABLE pedidos ENABLE ROW LEVEL SECURITY; CREATE POLICY "select proprios" ON pedidos FOR SELECT USING (auth.uid() = user_id); CREATE POLICY "insert proprios" ON pedidos FOR INSERT WITH CHECK (auth.uid() = user_id); CREATE POLICY "update proprios" ON pedidos FOR UPDATE USING (auth.uid() = user_id) WITH CHECK (auth.uid() = user_id); CREATE POLICY "delete proprios" ON pedidos FOR DELETE USING (auth.uid() = user_id);
